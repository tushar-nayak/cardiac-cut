# Pinned dependencies for the vascular cut tool, the authoritative list.
# These exact versions are the ones the tool was developed and verified against
# (Python 3.11). Every one ships prebuilt wheels for Windows / macOS / Linux on
# Python 3.10–3.12, so `pip install -r requirements.txt` needs no compiler.
#
# Python 3.13+ is NOT yet supported: several of these have no wheels there and
# would try (and fail) to build from source. Use Python 3.10, 3.11, or 3.12.

scipy==1.17.1
pandas==3.0.3
trimesh==4.12.2
shapely==2.1.2          # trimesh needs this for planar slicing (the cut itself)

pyvista==0.48.4         # 3-D rendering (pulls in VTK)
